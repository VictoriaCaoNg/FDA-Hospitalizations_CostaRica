lag_mat <- function(M, k = 2) {
  v <- as.vector(t(M))                    # one continuous weekly series
  v_lag <- c(rep(NA, k), head(v, -k))
  out <- matrix(v_lag, nrow = nrow(M), ncol = ncol(M), byrow = TRUE)
  dimnames(out) <- dimnames(M)
  out
}

lag_all <- function(base, k = 2, exclude = "Y", suffix = paste0("_lag", k)) {
  for (v in setdiff(names(base), exclude)) {
    x <- base[[v]]
    if (is.matrix(x)) {
      lagged <- lag_mat(unclass(x), k)    # unclass() strips AsIs if present
      base[[paste0(v, suffix)]] <- if (is.data.frame(base)) I(lagged) else lagged
    }
  }
  base
}

boot$ci_meta     # B_used, n_failed, failure_rate, levels, method
names(boot$smterms)
head(boot$smterms[[1]]$coef)

cf    <- coef(mod_cl2, sandwich = "cl2")   # same default grids as the bootstrap
tcrit <- qt(0.975, df = 19)                # G - 1 df

comp <- lapply(names(boot$smterms), function(nm) {
  b  <- boot$smterms[[nm]]$coef
  s  <- cf$smterms[[nm]]$coef
  bb <- boot$boot_ci$smterms[[nm]]
  stopifnot(nrow(b) == nrow(s))
  xcols <- setdiff(names(b), c("value", "se", "lower", "upper"))
  data.frame(b[, xcols, drop = FALSE],
             est     = b$value,
             cl2_lo  = s$value - tcrit * s$se,
             cl2_hi  = s$value + tcrit * s$se,
             boot_lo = bb[, "lower_95"],
             boot_hi = bb[, "upper_95"])
})
names(comp) <- names(boot$smterms)

summ <- do.call(rbind, lapply(names(comp), function(nm) {
  d <- comp[[nm]]
  sig_cl2  <- d$cl2_lo  > 0 | d$cl2_hi  < 0
  sig_boot <- d$boot_lo > 0 | d$boot_hi < 0
  data.frame(term = nm,
             width_ratio   = median((d$boot_hi - d$boot_lo) / (d$cl2_hi - d$cl2_lo)),
             pct_sig_cl2   = 100 * mean(sig_cl2),
             pct_sig_boot  = 100 * mean(sig_boot),
             pct_agreement = 100 * mean(sig_cl2 == sig_boot))
}))
print(summ, digits = 3)


library(ggplot2)
plot_term <- function(d, title) {
  x <- names(d)[1]                     # first coordinate column (lag/week index)
  ggplot(d, aes(.data[[x]], est)) +
    geom_ribbon(aes(ymin = boot_lo, ymax = boot_hi), fill = "grey70", alpha = .5) +
    geom_line(aes(y = cl2_lo), linetype = "dashed") +
    geom_line(aes(y = cl2_hi), linetype = "dashed") +
    geom_line(linewidth = .8) +
    geom_hline(yintercept = 0, colour = "grey40") +
    labs(title = title, x = x, y = "Coefficient",
         caption = "Grey: curve bootstrap 95%; dashed: CL2 (t, 19 df) 95%")
}
plots <- Map(plot_term, comp, names(comp))
